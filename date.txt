        class RhynoDatePicker {
            constructor(containerId) {
                this.container = document.getElementById(containerId);
                this.input = document.getElementById('dateInput');
                this.openBtn = document.getElementById('openBtn');
                this.window = document.getElementById('calendarWindow');
                this.titleBar = document.getElementById('titleBar');
                this.closeBtn = document.getElementById('closeBtn');
                this.headerTitle = document.getElementById('headerTitle');
                this.daysGrid = document.getElementById('daysGrid');

                this.prevYear = document.getElementById('prevYear');
                this.prevMonth = document.getElementById('prevMonth');
                this.nextMonth = document.getElementById('nextMonth');
                this.nextYear = document.getElementById('nextYear');
                this.todayBtn = document.getElementById('todayBtn');

                const today = this.gregorianToJalali(new Date());
                this.currentYear = today.year;
                this.currentMonth = today.month;
                this.selectedDateStr = '';

                this.initEvents();
                this.initDragAndDrop();
            }

            initEvents() {
                this.openBtn.addEventListener('click', (e) => {
                    e.stopPropagation();
                    this.window.classList.toggle('hidden');
                    if (!this.window.classList.contains('hidden')) {
                        this.render();
                    }
                });

                this.closeBtn.addEventListener('click', () => {
                    this.window.classList.add('hidden');
                });

                this.prevYear.addEventListener('click', () => this.changeYear(-1));
                this.nextYear.addEventListener('click', () => this.changeYear(1));
                this.prevMonth.addEventListener('click', () => this.changeMonth(-1));
                this.nextMonth.addEventListener('click', () => this.changeMonth(1));

                this.todayBtn.addEventListener('click', () => {
                    const today = this.gregorianToJalali(new Date());
                    this.currentYear = today.year;
                    this.currentMonth = today.month;
                    this.selectDate(today.year, today.month, today.day);
                });

                document.addEventListener('click', (e) => {
                    if (!this.container.contains(e.target)) {
                        this.window.classList.add('hidden');
                    }
                });
            }

            changeYear(delta) {
                this.currentYear += delta;
                this.render();
            }

            changeMonth(delta) {
                this.currentMonth += delta;
                if (this.currentMonth > 12) {
                    this.currentMonth = 1;
                    this.currentYear++;
                } else if (this.currentMonth < 1) {
                    this.currentMonth = 12;
                    this.currentYear--;
                }
                this.render();
            }

            toPersianDigits(num) {
                return String(num).replace(/\d/g, d => '۰۱۲۳۴۵۶۷۸۹'[d]);
            }

            render() {
                const monthNames = ['فروردین', 'اردیبهشت', 'خرداد', 'تیر', 'مرداد', 'شهریور', 'مهر', 'آبان', 'آذر', 'دی', 'بهمن', 'اسفند'];
                this.headerTitle.textContent = `${monthNames[this.currentMonth - 1]} ${this.toPersianDigits(this.currentYear)}`;

                this.daysGrid.innerHTML = '';
                const daysInMonth = this.getDaysInMonth(this.currentYear, this.currentMonth);
                const startDay = this.getFirstDayOfWeek(this.currentYear, this.currentMonth);

                for (let i = 0; i < startDay; i++) {
                    const empty = document.createElement('div');
                    empty.className = 'rhyno-day-cell empty';
                    this.daysGrid.appendChild(empty);
                }

                for (let d = 1; d <= daysInMonth; d++) {
                    const cell = document.createElement('div');
                    cell.className = 'rhyno-day-cell';
                    cell.textContent = this.toPersianDigits(d);

                    const dateStr = `${this.currentYear}/${String(this.currentMonth).padStart(2, '0')}/${String(d).padStart(2, '0')}`;
                    if (this.selectedDateStr === dateStr) {
                        cell.classList.add('selected');
                    }

                    cell.addEventListener('click', () => {
                        this.selectDate(this.currentYear, this.currentMonth, d);
                    });

                    this.daysGrid.appendChild(cell);
                }
            }

            selectDate(y, m, d) {
                this.selectedDateStr = `${y}/${String(m).padStart(2, '0')}/${String(d).padStart(2, '0')}`;
                this.input.value = this.toPersianDigits(this.selectedDateStr);
                this.render();
                this.window.classList.add('hidden');
            }

            initDragAndDrop() {
                let isDragging = false;
                let offsetX = 0;
                let offsetY = 0;

                this.titleBar.addEventListener('mousedown', (e) => {
                    if (e.target === this.closeBtn) return;
                    isDragging = true;
                    offsetX = e.clientX - this.window.offsetLeft;
                    offsetY = e.clientY - this.window.offsetTop;
                });

                document.addEventListener('mousemove', (e) => {
                    if (!isDragging) return;
                    this.window.style.left = (e.clientX - offsetX) + 'px';
                    this.window.style.top = (e.clientY - offsetY) + 'px';
                    this.window.style.right = 'auto';
                });

                document.addEventListener('mouseup', () => {
                    isDragging = false;
                });
            }

            getDaysInMonth(year, month) {
                if (month <= 6) return 31;
                if (month <= 11) return 30;
                return this.isLeapYear(year) ? 30 : 29;
            }

            isLeapYear(year) {
                const breaks = [-61, 9, 38, 199, 426, 686, 756, 818, 1111, 1181, 1210, 1635, 2060, 2097, 2192, 2262, 2324, 2394, 2456, 3178];
                let bl = breaks.length, jp = breaks[0], jm, jump, leap, n, i;
                if (year < jp || year >= breaks[bl - 1]) return false;
                for (i = 1; i < bl; i += 1) {
                    jm = breaks[i];
                    jump = jm - jp;
                    if (year < jm) break;
                    jp = jm;
                }
                n = year - jp;
                if (jump - n < 6) n = n - jump + (Math.floor((jump + 4) / 33) * 33);
                leap = ((((n + 1) % 33) - 1) % 4);
                if (leap === -1) leap = 4;
                return leap === 0;
            }

            getFirstDayOfWeek(year, month) {
                const gDate = this.jalaliToGregorian(year, month, 1);
                const day = new Date(gDate.year, gDate.month - 1, gDate.day).getDay();
                return (day + 2) % 7;
            }

            gregorianToJalali(gDate) {
                const gy = gDate.getFullYear();
                const gm = gDate.getMonth() + 1;
                const gd = gDate.getDate();
                const g_d_m = [0, 31, 59, 90, 120, 151, 181, 212, 243, 273, 304, 334];
                let jy = (gy <= 1600) ? 0 : 979;
                let gy2 = (gy <= 1600) ? gy - 621 : gy - 1600;
                let days = (365 * gy2) + (Math.floor(gy2 / 33) * 8) + Math.floor(((gy2 % 33) + 3) / 4) + 78 + gd + ((gm < 3) ? g_d_m[gm - 1] : g_d_m[gm - 1] + (((gy % 4 === 0 && gy % 100 !== 0) || (gy % 400 === 0)) ? 1 : 0));
                jy += 33 * Math.floor(days / 12053);
                days %= 12053;
                jy += 4 * Math.floor(days / 1461);
                days %= 1461;
                jy += Math.floor((days - 1) / 365);
                if (days > 0) days = (days - 1) % 365;
                let jm = (days < 186) ? 1 + Math.floor(days / 31) : 7 + Math.floor((days - 186) / 30);
                let jd = 1 + ((days < 186) ? (days % 31) : ((days - 186) % 30));
                return { year: jy, month: jm, day: jd };
            }

            jalaliToGregorian(jy, jm, jd) {
                let sal_a, gy, gm, gd, days;
                jy += 1595;
                days = -355668 + (365 * jy) + (Math.floor(jy / 33) * 8) + Math.floor(((jy % 33) + 3) / 4) + jd + ((jm < 7) ? (jm - 1) * 31 : ((jm - 7) * 30) + 186);
                gy = 400 * Math.floor(days / 146097);
                days %= 146097;
                if (days > 36524) {
                    gy += 100 * Math.floor(--days / 36524);
                    days %= 36524;
                    if (days >= 365) days++;
                }
                gy += 4 * Math.floor(days / 1461);
                days %= 1461;
                if (days > 365) {
                    gy += Math.floor((days - 1) / 365);
                    days = (days - 1) % 365;
                }
                gd = days + 1;
                sal_a = [0, 31, ((gy % 4 === 0 && gy % 100 !== 0) || (gy % 400 === 0)) ? 29 : 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31];
                for (gm = 0; gm < 13 && gd > sal_a[gm]; gm++) gd -= sal_a[gm];
                return { year: gy, month: gm, day: gd };
            }
        }

            new RhynoDatePicker('rhynoDatePicker');
        });
